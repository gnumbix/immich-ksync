# One QTest binary per macOS test suite, so a failure names the area it came from.
#
# Hermetic tests need no display, no session bus, no server and no network; they are
# the ones that run in CI and on every build.

set(IMMICHKSYNC_HERMETIC_TESTS
    ChecksumTest
    FolderAndFileNamingTest
    ReconciliationTest
    AlbumStructureTest
    SafetyGateTest
    TaskPoolTest
    LocalStateStoreTest
    FileSystemTest
    RetryTest
    ImmichErrorTest
    TlsCertificateStoreTest
    ImmichWireFormatTest
    SyncEngineTest
    TransportTest
    SecretStoreTest
)

foreach(test ${IMMICHKSYNC_HERMETIC_TESTS})
    add_executable(${test} ${test}.cpp)
    target_link_libraries(${test} PRIVATE immichksync_core Qt6::Test)
    target_include_directories(${test} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    # The certificate fixtures are generated by tools/make-test-certificates.sh and
    # committed, so the hermetic suite needs no openssl at build time.
    target_compile_definitions(${test} PRIVATE
        IMMICHKSYNC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
    # The secret store marshals D-Bus types, and asserting they are registered means
    # linking QtDBus into its suite.
    target_link_libraries(${test} PRIVATE Qt6::DBus Qt6::Network)
    add_test(NAME ${test} COMMAND ${test})
    # QStandardPaths::setTestModeEnabled keeps config and data out of the real home
    # directory, but the offscreen platform is what lets these run over SSH and in CI.
    #
    # The timeout matters: the bug these suites exist to catch makes a request wait for
    # ever rather than fail, so without it a regression hangs CI instead of reporting.
    set_tests_properties(${test} PROPERTIES
        ENVIRONMENT "QT_QPA_PLATFORM=offscreen"
        TIMEOUT 120)
endforeach()

# The interface tests link the presentation layer as well, and need a QGuiApplication.
# They run headless under the offscreen platform, so CI needs no display.
add_executable(InterfaceRenderTest
    InterfaceRenderTest.cpp
    ${CMAKE_SOURCE_DIR}/src/app/AppInfo.cpp
    ${CMAKE_SOURCE_DIR}/src/app/AppEnvironment.cpp
    ${CMAKE_SOURCE_DIR}/src/app/SystemEvents.cpp
    ${CMAKE_SOURCE_DIR}/src/settings/SettingsWindow.cpp
    ${CMAKE_SOURCE_DIR}/src/settings/SettingsWidgets.cpp
    ${CMAKE_SOURCE_DIR}/src/settings/ServerSettingsTab.cpp
    ${CMAKE_SOURCE_DIR}/src/settings/FolderSettingsTab.cpp
    ${CMAKE_SOURCE_DIR}/src/settings/AlbumsSettingsTab.cpp
    ${CMAKE_SOURCE_DIR}/src/settings/AdvancedSettingsTab.cpp
    ${CMAKE_SOURCE_DIR}/src/settings/LogTailWidget.cpp
)
target_link_libraries(InterfaceRenderTest PRIVATE
    immichksync_core Qt6::Test Qt6::Widgets
    KF6::ConfigWidgets KF6::WindowSystem KF6::IconThemes KF6::KIOGui)
target_include_directories(InterfaceRenderTest PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME InterfaceRenderTest COMMAND InterfaceRenderTest)
set_tests_properties(InterfaceRenderTest PROPERTIES
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen"
    TIMEOUT 120)

# Opt-in suites. Both are registered unconditionally and skip themselves when their
# environment variable is unset, so `ctest` stays green on a machine with neither a
# server nor the TLS harness running.
foreach(test TlsLiveTest LiveServerTest)
    add_executable(${test} ${test}.cpp)
    target_link_libraries(${test} PRIVATE immichksync_core Qt6::Test)
    target_include_directories(${test} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(${test} PRIVATE
        IMMICHKSYNC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
    add_test(NAME ${test} COMMAND ${test})
    set_tests_properties(${test} PROPERTIES
        ENVIRONMENT "QT_QPA_PLATFORM=offscreen"
        TIMEOUT 300)
endforeach()
